##### DQ

function dq(program)
{
    const stack = [];
    const randomisedProgram = program.split("").sort(() => Math.random() - 0.5);
    for(const character of randomisedProgram)
    {
        switch(character)
        {
            case "0":
                stack.length = 0;
                break;
            case "1":
                stack.push(1);
                break;
            case "D":
                if(stack.length > 0)
                    stack.push(stack[stack.length - 1]);
                break;
            case "Q":
                stack.pop();
                break;
        }
    }
    console.log(...stack);
}
