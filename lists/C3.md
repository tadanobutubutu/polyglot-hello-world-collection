   void matchbrackets()
   {
       int stack[MAXCODESIZE];
       int stackp = 0;
       
       for(codep=0; codep<codelength; codep++)

               case '{':
               case '[':
                   stack[stackp++]=codep; break;
               case '}':
               case ']':

                       --stackp;
